  function svgIcon(inner){
    return '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.6" stroke-linecap="round" stroke-linejoin="round">'+(inner||'')+'</svg>';
  }

  const TIER_WORD = {1:"YELLOW",2:"GREEN",3:"RED"};
  const TIER_CLASS = {1:"yellow",2:"green",3:"red"};

  const state = { role:"survivor", main:null, perks:[null,null,null] };

  const els = {
    body: document.body,
    roleBtns: document.querySelectorAll(".role-btn"),

    perkName:  [0,1,2].map(i=>document.getElementById("perkName"+i)),

    rollBtn: document.getElementById("rollBtn"),
    slipCode: document.getElementById("slipCode"),
    footerRole: document.getElementById("footerRole")
  };

  function bump(el){
    if(!el) return;
    el.classList.remove("is-rolling");
    void el.offsetWidth;
    el.classList.add("is-rolling");
  }

  function pickOne(arr, exclude){
    const pool = arr.filter(x=>!exclude.includes(x.name));
    return pool[Math.floor(Math.random()*pool.length)];
  }

  function pickN(arr, n){
    const shuffled = [...arr].sort(()=>Math.random()-0.5);
    return shuffled.slice(0,n);
  }

  function rollTier(){ return 1+Math.floor(Math.random()*3); }

  function tierTagHTML(t){
    return '<span class="tier-tag" data-tier="'+TIER_CLASS[t]+'"><i class="tier-dot"></i>'+TIER_WORD[t]+'</span>';
  }

  function newSlip(){
    const chars="23456789ABCDEFGHJKLMNPQRSTUVWXYZ";
    let s="VD-";
    for(let i=0;i<6;i++) s+=chars[Math.floor(Math.random()*chars.length)];
    return "SLIP #"+s;
  }

  const ALL_PERSONAL_PERKS = Object.keys(PERSONAL_PERKS).flatMap(code => PERSONAL_PERKS[code]);
  const ALL_KILLER_PERKS = KILLER_PERKS.concat(ALL_PERSONAL_PERKS);

  function mainPool(){
    return state.role==="survivor" ? ITEMS : KILLERS;
  }

  function perkPool(){
    return state.role==="survivor" ? SURVIVOR_PERKS : ALL_KILLER_PERKS;
  }

  function setMain(item, animate){
    state.main = item;
    els.mainIconTile.innerHTML = svgIcon(ICONS[item.code]);
    els.mainCode.textContent = item.code;
    els.mainName.textContent = item.name;
    els.mainDesc.textContent = item.desc;
    els.mainKicker.textContent = state.role==="survivor" ? "ITEM" : "KILLER";
    if(animate) bump(els.mainSlot);
  }

  function setPerk(i, perk, t, animate){
    state.perks[i] = Object.assign({}, perk, {tier:t});
    els.perkIconTile[i].innerHTML = svgIcon(ICONS[perk.cat]);
    els.perkCat[i].textContent = perk.kicker || perk.cat;
    els.perkName[i].textContent = perk.name;
    els.perkDesc[i].textContent = perk.desc;
    els.perkTier[i].innerHTML = tierTagHTML(t);
    if(animate) bump(els.perkCards[i]);
  }

  function rollAll(){
    setMain(pickOne(mainPool(), []), true);
    pickN(perkPool(), 3).forEach((perk,i)=> setPerk(i, perk, rollTier(), true));
    els.slipCode.textContent = newSlip();
  }

  function rerollMain(){
    setMain(pickOne(mainPool(), [state.main.name]), true);
  }

  function rerollPerk(i){
    const pool = perkPool();
    const others = state.perks.filter((_,idx)=>idx!==i).map(pk=>pk.name);
    const next = pickOne(pool, others.concat([state.perks[i].name]));
    setPerk(i, next, rollTier(), true);
  }

  function setRole(role){
    if(role===state.role) return;
    state.role = role;
    els.body.dataset.role = role;
    els.roleBtns.forEach(btn=>{
      const active = btn.dataset.roleBtn===role;
      btn.classList.toggle("is-active", active);
      btn.setAttribute("aria-pressed", String(active));
    });
    els.footerRole.textContent = role==="survivor" ? "SURVIVOR BUILD" : "KILLER BUILD";
    rollAll();
  }

  els.roleBtns.forEach(btn=> btn.addEventListener("click", ()=> setRole(btn.dataset.roleBtn)));
  els.rollBtn.addEventListener("click", rollAll);
  els.mainSlot.addEventListener("click", rerollMain);
  els.perkCards.forEach((card,i)=> card.addEventListener("click", ()=> rerollPerk(i)));

  // Initial Run
  rollAll();
});
